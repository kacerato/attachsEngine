cmake_minimum_required(VERSION 3.22)
project(AstraPostCacheGpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(ASTRA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." CACHE PATH "Astra source root")
find_path(VULKAN_HEADERS vulkan/vulkan.h REQUIRED)
find_program(GLSLC glslc REQUIRED)
foreach(STAGE vert frag)
  if(STAGE STREQUAL "vert")
    set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/cache.vert")
    set(OUTPUT "cache.vert.spv")
  else()
    set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/ui.frag")
    set(OUTPUT "ui.frag.spv")
  endif()
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${OUTPUT}" COMMAND "${GLSLC}" "${SOURCE}" -o "${CMAKE_CURRENT_BINARY_DIR}/${OUTPUT}" DEPENDS "${SOURCE}")
  list(APPEND SHADERS "${CMAKE_CURRENT_BINARY_DIR}/${OUTPUT}")
endforeach()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/post.frag.spv" COMMAND "${GLSLC}" -I "${ASTRA_ROOT}/native/rhi/shaders" "${ASTRA_ROOT}/native/rhi/shaders/post_process.frag" -o "${CMAKE_CURRENT_BINARY_DIR}/post.frag.spv" DEPENDS "${ASTRA_ROOT}/native/rhi/shaders/post_process.frag")
add_executable(astra_post_cache_gpu main.cpp ${SHADERS} "${CMAKE_CURRENT_BINARY_DIR}/post.frag.spv")
file(COPY "${VULKAN_HEADERS}/vulkan" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include")
if(EXISTS "${VULKAN_HEADERS}/vk_video")
 file(COPY "${VULKAN_HEADERS}/vk_video" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include")
endif()
target_include_directories(astra_post_cache_gpu PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/include" "${ASTRA_ROOT}/native")
if(UNIX)
 target_link_libraries(astra_post_cache_gpu PRIVATE dl)
endif()
enable_testing()
add_test(NAME post_cache_pixels COMMAND astra_post_cache_gpu "${CMAKE_CURRENT_BINARY_DIR}")
